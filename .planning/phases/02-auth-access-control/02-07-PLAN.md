Create a new migration file `supabase/migrations/20260324_update_profiles_trigger_phone.sql` to override `public.handle_new_user()` so it reads the phone from `raw_user_meta_data` (where Register.tsx now puts it) and falls back to `NEW.phone`.

```sql
-- Update trigger to extract phone from raw_user_meta_data 
-- since we are using dummy emails instead of native phone auth

CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  INSERT INTO public.profiles (id, full_name, phone, year_of_birth, address)
  VALUES (
    NEW.id,
    COALESCE(NEW.raw_user_meta_data->>'full_name', ''),
    COALESCE(NEW.raw_user_meta_data->>'phone', NEW.phone, ''),
    COALESCE((NEW.raw_user_meta_data->>'year_of_birth')::INTEGER, 0),
    COALESCE(NEW.raw_user_meta_data->>'address', '')
  );
  RETURN NEW;
END;
$$;
```
