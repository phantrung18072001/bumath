```typescript
import { z } from 'zod'
import { isValidVnPhone } from '@/lib/validators'

const registerSchema = z.object({
  phone: z.string().refine(isValidVnPhone, { message: 'Số điện thoại không hợp lệ (VD: 0912345678)' }),
  fullName: z.string().min(1, 'Vui lòng nhập tên học sinh'),
  yearOfBirth: z.coerce.number().min(1990, 'Năm sinh không hợp lệ').max(2020, 'Năm sinh không hợp lệ'),
  address: z.string().min(1, 'Vui lòng nhập địa chỉ'),
  password: z.string().min(8, 'Mật khẩu phải có ít nhất 8 ký tự.'),
  confirmPassword: z.string(),
}).refine((data) => data.password === data.confirmPassword, {
  message: 'Mật khẩu xác nhận không khớp.',
  path: ['confirmPassword'],
})
```
