      <span>Hiển thị</span>

        <SelectTrigger className="w-[72px] h-8">
          <SelectValue />

      <span>/ trang</span>

      <Pagination>
        <PaginationContent>
          <PaginationItem>
            <PaginationPrevious
              onClick={() => setCurrentPage((p) => Math.max(1, p - 1))}
              aria-disabled={currentPage === 1}

          </PaginationItem>

                <PaginationLink

                >
                  {page}
                </PaginationLink>
              </PaginationItem>
            )
          )}
          <PaginationItem>
            <PaginationNext

        </PaginationContent>
      </Pagination>

  </div>
